// ============================================================ // VIVES SOLUCIONES — Portal de Clientes v2 // Google Apps Script — Backend completo // Pegá este código en: Extensiones > Apps Script // ============================================================ function doGet(e) { var params = e.parameter; var action = params.action || ''; var cid = params.cliente_id || ''; var result; if (action === 'login') result = login(params.cuit, params.password); else if (action === 'pedidos') result = getPedidos(cid, params.rubro); else if (action === 'facturas') result = getFacturas(cid, params.rubro); else if (action === 'vencimientos') result = getVencimientos(cid, params.rubro); else if (action === 'matafuegos') result = getMatafuegos(cid); else if (action === 'presupuestos') result = getPresupuestos(cid, params.rubro); else if (action === 'ordenes_compra') result = getOrdenesCompra(cid, params.rubro); else if (action === 'comprobantes') result = getComprobantes(cid); else if (action === 'resumen') result = getResumen(cid); else if (action === 'rubros') result = getRubros(cid); else result = { ok: false, error: 'Acción no reconocida' }; return resp(result); } // ── HELPERS ────────────────────────────────────────────────── function hoja(nombre) { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(nombre); } function toObj(datos, desde) { var cab = datos[0]; var result = []; for (var i = desde || 1; i < datos.length; i++) { if (datos[i].every(function(c){ return c === '' || c === null; })) continue; var obj = {}; for (var j = 0; j < cab.length; j++) { var v = datos[i][j]; obj[cab[j]] = v instanceof Date ? Utilities.formatDate(v, Session.getScriptTimeZone(), 'dd/MM/yyyy') : v; } result.push(obj); } return result; } function filtrar(rows, campo, valor) { return rows.filter(function(r) { return String(r[campo] || '').trim() === String(valor).trim(); }); } function calcEstadoVenc(fechaStr) { if (!fechaStr) return { estado: '—', dias: null }; var partes = fechaStr.split('/'); if (partes.length < 3) return { estado: '—', dias: null }; var fecha = new Date(partes[2], partes[1]-1, partes[0]); var hoy = new Date(); hoy.setHours(0,0,0,0); var dias = Math.round((fecha - hoy) / 86400000); var estado = dias < 0 ? 'Vencido' : dias <= 7 ? 'Próximo' : 'Al día'; return { estado: estado, dias: dias }; } function resp(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ── LOGIN (por CUIT) ────────────────────────────────────────── function login(cuit, password) { if (!cuit || !password) return { ok: false, error: 'Faltan credenciales' }; var h = hoja('Clientes'); if (!h) return { ok: false, error: 'Hoja Clientes no encontrada' }; var rows = toObj(h.getDataRange().getValues()); for (var i = 0; i < rows.length; i++) { var r = rows[i]; var cuitHoja = String(r.cuit || '').replace(/-/g, '').trim(); var cuitInput = String(cuit || '').replace(/-/g, '').trim(); var passHoja = String(r.password || '').trim(); if (cuitHoja === cuitInput && passHoja === password.trim()) { if (String(r.activo || '').toUpperCase() !== 'SI') { return { ok: false, error: 'Cuenta inactiva. Contactá a Vives Soluciones.' }; } var rubros = String(r.rubros_activos || '').split('|').map(function(s){ return s.trim(); }).filter(Boolean); return { ok: true, cliente_id: r.cliente_id, razon_social: r.razon_social, cuit: r.cuit, contacto: r.contacto, sector: r.sector, rubros: rubros }; } } return { ok: false, error: 'CUIT o contraseña incorrectos' }; } // ── RUBROS DEL CLIENTE ──────────────────────────────────────── function getRubros(cid) { if (!cid) return { ok: false, error: 'cliente_id requerido' }; var h = hoja('Clientes'); if (!h) return { ok: false, error: 'Hoja no encontrada' }; var rows = toObj(h.getDataRange().getValues()); var cliente = rows.find(function(r){ return r.cliente_id === cid; }); if (!cliente) return { ok: false, error: 'Cliente no encontrado' }; var rubros = String(cliente.rubros_activos || '').split('|').map(function(s){ return s.trim(); }).filter(Boolean); return { ok: true, rubros: rubros }; } // ── PEDIDOS ─────────────────────────────────────────────────── function getPedidos(cid, rubro) { if (!cid) return { ok: false, error: 'cliente_id requerido' }; var h = hoja('Pedidos'); if (!h) return { ok: false, error: 'Hoja Pedidos no encontrada' }; var rows = filtrar(toObj(h.getDataRange().getValues()), 'cliente_id', cid); if (rubro) rows = filtrar(rows, 'rubro', rubro); return { ok: true, datos: rows }; } // ── FACTURAS ────────────────────────────────────────────────── function getFacturas(cid, rubro) { if (!cid) return { ok: false, error: 'cliente_id requerido' }; var h = hoja('Facturas'); if (!h) return { ok: false, error: 'Hoja Facturas no encontrada' }; var rows = filtrar(toObj(h.getDataRange().getValues()), 'cliente_id', cid); if (rubro) rows = filtrar(rows, 'rubro', rubro); return { ok: true, datos: rows }; } // ── VENCIMIENTOS (con cálculo automático) ───────────────────── function getVencimientos(cid, rubro) { if (!cid) return { ok: false, error: 'cliente_id requerido' }; var h = hoja('Vencimientos'); if (!h) return { ok: false, error: 'Hoja Vencimientos no encontrada' }; var rows = filtrar(toObj(h.getDataRange().getValues()), 'cliente_id', cid); if (rubro) rows = filtrar(rows, 'rubro', rubro); rows = rows.map(function(r) { var calc = calcEstadoVenc(r.fecha_vencimiento); r.estado = calc.estado; r.dias_restantes = calc.dias; return r; }); rows.sort(function(a, b){ return (a.dias_restantes || 0) - (b.dias_restantes || 0); }); return { ok: true, datos: rows }; } // ── MATAFUEGOS ──────────────────────────────────────────────── function getMatafuegos(cid) { if (!cid) return { ok: false, error: 'cliente_id requerido' }; var h = hoja('Matafuegos'); if (!h) return { ok: false, error: 'Hoja Matafuegos no encontrada' }; var rows = filtrar(toObj(h.getDataRange().getValues()), 'cliente_id', cid); rows = rows.map(function(r) { var calc = calcEstadoVenc(r.fecha_vencimiento); r.estado = calc.estado; r.dias_restantes = calc.dias; return r; }); rows.sort(function(a, b){ return (a.dias_restantes || 0) - (b.dias_restantes || 0); }); return { ok: true, datos: rows }; } // ── PRESUPUESTOS ────────────────────────────────────────────── function getPresupuestos(cid, rubro) { if (!cid) return { ok: false, error: 'cliente_id requerido' }; var h = hoja('Presupuestos'); if (!h) return { ok: false, error: 'Hoja Presupuestos no encontrada' }; var rows = filtrar(toObj(h.getDataRange().getValues()), 'cliente_id', cid); if (rubro) rows = filtrar(rows, 'rubro', rubro); return { ok: true, datos: rows }; } // ── ORDENES DE COMPRA ───────────────────────────────────────── function getOrdenesCompra(cid, rubro) { if (!cid) return { ok: false, error: 'cliente_id requerido' }; var h = hoja('OrdenesCompra'); if (!h) return { ok: false, error: 'Hoja OrdenesCompra no encontrada' }; var rows = filtrar(toObj(h.getDataRange().getValues()), 'cliente_id', cid); if (rubro) rows = filtrar(rows, 'rubro', rubro); return { ok: true, datos: rows }; } // ── COMPROBANTES DE PAGO ────────────────────────────────────── function getComprobantes(cid) { if (!cid) return { ok: false, error: 'cliente_id requerido' }; var h = hoja('ComprobantesPago'); if (!h) return { ok: false, error: 'Hoja ComprobantesPago no encontrada' }; var rows = filtrar(toObj(h.getDataRange().getValues()), 'cliente_id', cid); return { ok: true, datos: rows }; } // ── RESUMEN GENERAL ─────────────────────────────────────────── function getResumen(cid) { if (!cid) return { ok: false, error: 'cliente_id requerido' }; var hoy = new Date(); hoy.setHours(0,0,0,0); // Pedidos activos var pedActivos = 0; var hP = hoja('Pedidos'); if (hP) { filtrar(toObj(hP.getDataRange().getValues()), 'cliente_id', cid).forEach(function(r) { if (r.estado !== 'Entregado' && r.estado !== '') pedActivos++; }); } // Alertas vencimiento (vencidos + próximos 7 días) var alertas = 0; var hV = hoja('Vencimientos'); if (hV) { filtrar(toObj(hV.getDataRange().getValues()), 'cliente_id', cid).forEach(function(r) { var d = calcEstadoVenc(r.fecha_vencimiento).dias; if (d !== null && d <= 7) alertas++; }); } // Matafuegos con alerta var matAlert = 0; var hM = hoja('Matafuegos'); if (hM) { filtrar(toObj(hM.getDataRange().getValues()), 'cliente_id', cid).forEach(function(r) { var d = calcEstadoVenc(r.fecha_vencimiento).dias; if (d !== null && d <= 30) matAlert++; }); } // Saldo pendiente (facturas) var saldo = 0; var hF = hoja('Facturas'); if (hF) { filtrar(toObj(hF.getDataRange().getValues()), 'cliente_id', cid).forEach(function(r) { if (r.estado === 'Pendiente' || r.estado === 'Vencida') { saldo += Number(r.importe) || 0; } }); } // OC pendientes var ocPend = 0; var hO = hoja('OrdenesCompra'); if (hO) { filtrar(toObj(hO.getDataRange().getValues()), 'cliente_id', cid).forEach(function(r) { if (r.estado === 'Pendiente') ocPend++; }); } return { ok: true, resumen: { pedidos_activos: pedActivos, alertas_venc: alertas, mat_alertas: matAlert, saldo_pendiente: saldo, oc_pendientes: ocPend } }; } // ── ADMIN: LISTAR CLIENTES ──────────────────────────────────── function admin_clientes() { var h = hoja('Clientes'); if (!h) return { ok: false, error: 'Hoja no encontrada' }; var datos = h.getDataRange().getValues(); var cab = datos[0]; var result = []; for (var i = 1; i < datos.length; i++) { if (datos[i].every(function(c){ return c === '' || c === null; })) continue; var obj = { _rowIndex: i + 1 }; for (var j = 0; j < cab.length; j++) { obj[cab[j]] = datos[i][j]; } result.push(obj); } return { ok: true, datos: result }; } // ── ADMIN: NUEVO CLIENTE ────────────────────────────────────── function admin_nuevo_cliente(params) { var h = hoja('Clientes'); if (!h) return { ok: false, error: 'Hoja no encontrada' }; var datos = h.getDataRange().getValues(); var cab = datos[0]; // Generar cliente_id var maxId = 0; for (var i = 1; i < datos.length; i++) { var id = String(datos[i][0] || '').replace('CLI-', ''); if (!isNaN(id) && parseInt(id) > maxId) maxId = parseInt(id); } var nuevoId = 'CLI-' + String(maxId + 1).padStart(3, '0'); var fila = []; cab.forEach(function(c) { if (c === 'cliente_id') fila.push(nuevoId); else if (c === 'razon_social') fila.push(params.razon_social || ''); else if (c === 'cuit') fila.push(params.cuit || ''); else if (c === 'password') fila.push(params.password || ''); else if (c === 'contacto') fila.push(params.contacto || ''); else if (c === 'email') fila.push(params.email || ''); else if (c === 'telefono') fila.push(params.telefono || ''); else if (c === 'sector') fila.push(params.sector || ''); else if (c === 'activo') fila.push(params.activo || 'SI'); else if (c === 'rubros_activos') fila.push(params.rubros_activos || ''); else fila.push(''); }); h.appendRow(fila); return { ok: true, cliente_id: nuevoId }; } // ── ADMIN: EDITAR CLIENTE ───────────────────────────────────── function admin_editar_cliente(params) { var h = hoja('Clientes'); if (!h) return { ok: false, error: 'Hoja no encontrada' }; var rowIndex = parseInt(params.row_index); if (!rowIndex) return { ok: false, error: 'row_index requerido' }; var cab = h.getDataRange().getValues()[0]; cab.forEach(function(c, j) { var col = j + 1; if (c === 'razon_social' && params.razon_social) h.getRange(rowIndex, col).setValue(params.razon_social); else if (c === 'cuit' && params.cuit) h.getRange(rowIndex, col).setValue(params.cuit); else if (c === 'password' && params.password) h.getRange(rowIndex, col).setValue(params.password); else if (c === 'contacto') h.getRange(rowIndex, col).setValue(params.contacto || ''); else if (c === 'email') h.getRange(rowIndex, col).setValue(params.email || ''); else if (c === 'telefono') h.getRange(rowIndex, col).setValue(params.telefono || ''); else if (c === 'sector') h.getRange(rowIndex, col).setValue(params.sector || ''); else if (c === 'activo') h.getRange(rowIndex, col).setValue(params.activo || 'SI'); else if (c === 'rubros_activos') h.getRange(rowIndex, col).setValue(params.rubros_activos || ''); }); return { ok: true }; } // ── ADMIN: TOGGLE ACTIVO ────────────────────────────────────── function admin_toggle_activo(params) { var h = hoja('Clientes'); if (!h) return { ok: false, error: 'Hoja no encontrada' }; var rowIndex = parseInt(params.row_index); var cab = h.getDataRange().getValues()[0]; var colActivo = cab.indexOf('activo') + 1; if (!colActivo) return { ok: false, error: 'Columna activo no encontrada' }; h.getRange(rowIndex, colActivo).setValue(params.activo || 'SI'); return { ok: true }; }
